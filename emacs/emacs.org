#+title: Vanilla Emacs
#+property: header-args:emacs-lisp :tangle config.el :mkdirp yes

This is the test configuration for the separate vanilla Emacs daemon.
It intentionally stays smaller than Spacemacs while keeping the parts that make
daily editing usable: Evil, discoverable keys, modern minibuffer completion,
manual formatting, LSP through Eglot, and built-in tab-bar workspaces.

  (defconst my/home-buffer-name "*home*"
    "Name of the vanilla Emacs home buffer.")

  (defgroup my/home nil
    "Vanilla Emacs home buffer."
    :group 'convenience)

  (defface my/home-title
    '((t :inherit default :height 1.45 :weight regular))
    "Face for the home buffer title."
    :group 'my/home)

  (defface my/home-section
    '((t :inherit font-lock-keyword-face :height 1.05 :weight regular))
    "Face for home buffer section headings."
    :group 'my/home)

  (defface my/home-muted
    '((t :inherit shadow))
    "Face for secondary home buffer text."
    :group 'my/home)

  (defun my/project-root-for-path (path)
    "Return a project root for PATH, or nil."
    (when-let* ((directory (cond
                            ((file-directory-p path) path)
                            ((file-exists-p path) (file-name-directory path)))))
      (or (locate-dominating-file directory ".git")
          (locate-dominating-file directory "flake.nix"))))

  (defun my/recent-project-roots (&optional limit)
    "Return recent project roots, preferring recently visited files."
    (let ((roots nil))
      (dolist (file recentf-list)
        (when-let* ((root (my/project-root-for-path file)))
          (cl-pushnew (directory-file-name root) roots :test #'string=)))
      (when (fboundp 'project-known-project-roots)
        (dolist (root (project-known-project-roots))
          (when (file-directory-p root)
            (cl-pushnew (directory-file-name root) roots :test #'string=))))
      (setq roots (nreverse roots))
      (if limit
          (seq-take roots (min limit (length roots)))
        roots)))

  (defun my/home-insert-button (label action &optional help)
    "Insert LABEL as a home buffer button running ACTION."
    (insert-text-button
     label
     'follow-link t
     'help-echo help
     'action (lambda (_button) (funcall action))))

  (defun my/home-open-project (root)
    "Open ROOT as a folder tab."
    (my/open-directory-tab root))

  (defun my/home-refresh ()
    "Render the vanilla Emacs home buffer."
    (let ((inhibit-read-only t))
      (erase-buffer)
      (setq-local line-spacing 0.15)
      (insert "\n")
      (insert (propertize "Emacs" 'face 'my/home-title) "\n")
      (insert (propertize "Recent work, common entrypoints, and quiet defaults." 'face 'my/home-muted) "\n\n")

      (insert (propertize "Projects" 'face 'my/home-section) "\n")
      (let ((projects (my/recent-project-roots 8)))
        (if projects
            (dolist (root projects)
              (let ((label (file-name-nondirectory root)))
                (insert "  ")
                (my/home-insert-button
                 (if (string-empty-p label) root label)
                 (lambda () (my/home-open-project root))
                 root)
                (insert (propertize (format "  %s" root) 'face 'my/home-muted) "\n")))
          (insert (propertize "  No recent projects yet." 'face 'my/home-muted) "\n")))

      (insert "\n" (propertize "Files" 'face 'my/home-section) "\n")
      (let ((files (seq-take recentf-list (min 8 (length recentf-list)))))
        (if files
            (dolist (file files)
              (insert "  ")
              (my/home-insert-button
               (file-name-nondirectory file)
               (lambda () (my/find-file-or-directory file))
               file)
              (insert (propertize (format "  %s" (abbreviate-file-name file)) 'face 'my/home-muted) "\n"))
          (insert (propertize "  No recent files yet." 'face 'my/home-muted) "\n")))

      (insert "\n" (propertize "Actions" 'face 'my/home-section) "\n")
      (insert "  ")
      (my/home-insert-button "Find file" #'my/find-file-or-directory)
      (insert "    ")
      (my/home-insert-button "Project file" #'project-find-file)
      (insert "    ")
      (my/home-insert-button "Search" #'consult-ripgrep)
      (insert "    ")
      (my/home-insert-button "Mail" #'my/mu4e-open-inbox)
      (insert "    ")
      (my/home-insert-button "Magit" #'magit-status)
      (insert "\n\n")
      (goto-char (point-min))))

  (defun my/home ()
    "Open the vanilla Emacs home buffer."
    (interactive)
    (let ((buffer (get-buffer-create my/home-buffer-name)))
      (with-current-buffer buffer
        (special-mode)
        (setq-local display-line-numbers nil
                    truncate-lines t)
        (let ((map (make-sparse-keymap)))
          (set-keymap-parent map special-mode-map)
          (define-key map (kbd "g") #'my/home-refresh)
          (define-key map (kbd "f") #'my/find-file-or-directory)
          (define-key map (kbd "p") #'project-switch-project)
          (define-key map (kbd "s") #'consult-ripgrep)
          (define-key map (kbd "m") #'my/mu4e-open-inbox)
          (use-local-map map))
        (my/home-refresh))
      (switch-to-buffer buffer)))

  (setq initial-buffer-choice #'my/home)

    "Return the tab assigned to workspace NUMBER."
    (seq-find (lambda (tab)
                (= (or (my/workspace-number tab) 0) number))
              (tab-bar-tabs)))

  (defun my/workspace-context-name ()
    "Return a concise context name for the current workspace."
    (cond
     ((and (derived-mode-p 'dired-mode) default-directory)
      (my/directory-tab-name default-directory))
     (buffer-file-name
      (my/directory-tab-name (file-name-directory buffer-file-name)))
     (t nil)))

  (defun my/refresh-workspace-label ()
    "Show the workspace number and current file or directory context."
    (let* ((tab (my/current-workspace-tab))
           (number (my/workspace-number tab))
           (context (my/workspace-context-name)))
      (when number
        (tab-rename (if context
                        (format "%d: %s" number context)
                      (number-to-string number))))))

  (defun my/assign-workspace-number (number)
    "Assign NUMBER to the current workspace and refresh its label."
    (let* ((tab (my/current-workspace-tab))
           (entry (assq 'my/workspace-number tab)))
      (if entry
          (setcdr entry number)
        (setcdr tab (cons (cons 'my/workspace-number number) (cdr tab)))))
    (my/refresh-workspace-label))

  (defun my/select-workspace (number)
