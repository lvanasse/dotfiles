This is the literate Spacemacs configuration for my NixOS dotfiles.  Each
section documents *why* a piece of configuration exists so that the intent
stays clear over time.

The Org file is tangled at Nix build time (=emacs --batch= with
=org-babel-tangle-file=) to produce the =.spacemacs= that Spacemacs loads.
Nix path substitution placeholders (=__MU_BIN__=, etc.) are replaced after
tangling, just as before.

A comment header so that the generated file identifies its origin.

* Layers

The =dotspacemacs/layers= function declares which Spacemacs layers to load,
additional packages that aren't part of any layer, and packages to exclude.

The layers are organized as:

- *Core*: =better-defaults=, =git=, =org= — standard productivity
- *Completion*: =auto-completion= (with snippets), =spell-checking=,
  =syntax-checking=, =ivy=, =lsp=, =treemacs=
- *Languages*: C/C++ (clangd), Claude Code, CMake, Docker, Markdown, Nix,
  PDF, Python (pyright), YAML, ERC (IRC), Rust, Shell, plus embedded-adjacent
  file formats (TOML, Meson, Device Tree, Kconfig, Lua)
- *Communication*: =mu4e= (email), =erc= (IRC), =slack=, =spotify=
- *Tools*: =tabs= (centaur-tabs), =shell=, =ghostel=, =shell-scripts=

Additional packages (=gruvbox-theme=, =ghostel=, =vterm=, =multi-vterm=, =clipetty=, etc.) are
installed outside layers.  =forge= is excluded to avoid pulling in heavy
Git forge dependencies.

* User Init

Runs early, before layers are loaded.  Used for performance tuning and
disabling UI chrome that would otherwise flash on startup.

** Performance tuning

Raise the GC threshold to maximum and disable =file-name-handler-alist=
during startup so that package initialization doesn't trigger excessive
garbage collection or regex matching.  After startup, =gcmh= takes over
GC management (collecting only during idle time).
=fast-but-imprecise-scrolling=, =redisplay-skip-fontification-on-input=,
and =jit-lock-defer-time= improve interactive responsiveness.
=inhibit-compacting-font-caches= prevents GC stalls from font cache
compaction in GUI frames.  Bidi scanning is disabled (LTR-only) and
=scroll-conservatively= avoids jarring recentering jumps.
=lsp-use-plists= enables plist-based LSP deserialization which is
significantly faster than the default hash-table mode.

** UI chrome removal

Disable menu bar, toolbar, scrollbar, blinking cursor, tooltips, fringes,
and tab-bar so the editor starts with a clean, distraction-free look.

** Native compilation warnings

Silence native-compilation async warnings that would otherwise pop up
buffers during normal editing after a fresh rebuild.

** Frame defaults

Remove the internal border so that tiling window managers (Sway) get a
seamless appearance.

#+begin_src emacs-lisp
(defun dotspacemacs/user-init ()
  ;; Bypass file-name-handler-alist during startup (restored after init).
  (defvar my/file-name-handler-alist-backup file-name-handler-alist)
  (setq file-name-handler-alist nil)
  (add-hook 'emacs-startup-hook
            (lambda ()
              (setq file-name-handler-alist my/file-name-handler-alist-backup)))

  ;; Use plist deserialization for lsp-mode (faster than hash-tables), but keep
  ;; language servers out of formatting decisions unless explicitly requested.
  (setenv "LSP_USE_PLISTS" "true")
  (setq lsp-use-plists t
        lsp-enable-indentation nil
        lsp-enable-on-type-formatting nil
        lsp-before-save-edits nil
        lsp-completion-provider :capf
        lsp-idle-delay 0.35
        lsp-pyright-langserver-command "basedpyright-langserver")
  (when (boundp 'treesit-extra-load-path)
    (add-to-list 'treesit-extra-load-path
                 (expand-file-name "tree-sitter" user-emacs-directory)))

  ;; Startup and LSP defaults tuned for interactive work rather than cold-start demos.
  (setq gc-cons-threshold most-positive-fixnum
        gc-cons-percentage 0.6
        read-process-output-max (* 4 1024 1024)
        idle-update-delay 1.0
        fast-but-imprecise-scrolling t
        redisplay-skip-fontification-on-input t
        inhibit-compacting-font-caches t
        site-run-file nil
        jit-lock-defer-time 0.05)

  ;; Avoid expensive bidi reordering scans (only LTR text in this setup).
  (setq-default bidi-display-reordering 'left-to-right
                bidi-paragraph-direction 'left-to-right)
  (setq bidi-inhibit-bpa t)

  ;; Smooth scrolling without recentering jumps.
  (setq scroll-conservatively 101
        scroll-margin 2
        auto-window-vscroll nil)
  (setq-default cursor-type '(hbar . 3)
                line-spacing 2
                indent-tabs-mode nil
                tab-width 4
                standard-indent 4
                evil-shift-width 4)
  (menu-bar-mode -1)
  (tool-bar-mode -1)
  (scroll-bar-mode -1)
  (blink-cursor-mode -1)
  (when (fboundp 'tooltip-mode)
    (tooltip-mode -1))
  (when (fboundp 'set-fringe-mode)
    (set-fringe-mode 4))
  (when (fboundp 'tab-bar-mode)
    (tab-bar-mode -1))
  ;; Emacs 31 can leave these Gnus faces in a reverse inheritance chain before
  ;; gruvbox loads.  Reset them immediately before every theme load so gruvbox
  ;; can install its own non-cyclic inheritance.
  (defun my/prepare-gruvbox-theme-faces (&rest _)
    (dolist (face '(gnus-group-news-low gnus-group-news-low-empty))
      (when (facep face)
        (set-face-attribute face nil :inherit nil))))
  (my/prepare-gruvbox-theme-faces)
  (advice-add 'load-theme :before #'my/prepare-gruvbox-theme-faces)
  (when (boundp 'native-comp-async-report-warnings-errors)
    (setq native-comp-async-report-warnings-errors 'silent))
  (when (boundp 'comp-async-report-warnings-errors)
    (setq comp-async-report-warnings-errors 'silent))
  ;; MELPA packages can be noisy under Emacs 31 while Spacemacs installs and
  ;; byte/native-compiles them during startup.  Keep the startup frame focused
  ;; on real load errors instead of third-party compiler lint.
  (setq byte-compile-warnings nil)
  (when (boundp 'native-comp-warning-on-missing-source)
    (setq native-comp-warning-on-missing-source nil))
  (when (boundp 'warning-suppress-types)
    (add-to-list 'warning-suppress-types '(native-compiler)))
  (when (boundp 'warning-suppress-log-types)
    (add-to-list 'warning-suppress-log-types '(native-compiler)))
  (require 'seq)
  ;; Client frames can fall back to a transient buffer with no Evil leader
  ;; bindings after killing the visited file. Replace that state with another
  ;; real editing buffer, or a normal scratch buffer as a last resort.
  (defun my/normalize-fallback-buffer (&optional buffer)
    (with-current-buffer (or buffer (current-buffer))
      (unless (eq major-mode 'text-mode)
        (text-mode))))

  (defun my/show-fallback-scratch-buffer ()
    (let ((buffer (get-buffer-create "*scratch*")))
      (switch-to-buffer buffer)
      (my/normalize-fallback-buffer buffer)
      buffer))

  (defun my/interactive-buffer-p (buffer)
    "Return non-nil when BUFFER is a sensible post-kill destination."
    (when (buffer-live-p buffer)
      (with-current-buffer buffer
        (or buffer-file-name
            (and (buffer-base-buffer)
                 (buffer-file-name (buffer-base-buffer)))
            (derived-mode-p 'dired-mode)
            (string-match-p
             "\\`\\*\\(?:terminal\\|ghostel\\|shell\\|eshell\\|vterm\\)"
             (buffer-name buffer))))))

  (defun my/post-kill-replacement-buffer (&optional killed-buffer)
    "Return a live buffer to focus after KILLED-BUFFER is gone."
    (seq-find (lambda (buffer)
                (and (not (eq buffer killed-buffer))
                     (my/interactive-buffer-p buffer)))
              (buffer-list)))

  (defun my/bad-fallback-buffer-p (buffer)
    (with-current-buffer buffer
      (or (not (my/interactive-buffer-p buffer))
          (equal (buffer-name buffer) " *server*")
          (and (equal (buffer-name buffer) "*scratch*")
               (eq major-mode 'fundamental-mode)))))

  (defun my/restore-input-state-after-buffer-kill ()
    "Reset transient input state after buffer deletion."
    (setq overriding-local-map nil
          overriding-terminal-local-map nil)
    (deactivate-mark)
    (when (and (fboundp 'evil-normal-state)
               (not (minibufferp)))
      (evil-normal-state)))

  (defun my/repair-window-after-buffer-kill (window killed-buffer)
    "Repair WINDOW when it lands on a bad buffer after KILLED-BUFFER is gone."
    (when (window-live-p window)
      (with-selected-window window
        (let ((buffer (window-buffer window)))
          (when (my/bad-fallback-buffer-p buffer)
            (let ((replacement (my/post-kill-replacement-buffer killed-buffer)))
              (if replacement
                  (switch-to-buffer replacement)
                (my/show-fallback-scratch-buffer)))
            (my/restore-input-state-after-buffer-kill))))))

  (defun my/repair-buffer-fallback-after-kill (&optional killed-buffer)
    (let ((frame (selected-frame)))
      (run-at-time
       0 nil
       (lambda (target-frame target-buffer)
         (when (frame-live-p target-frame)
           (with-selected-frame target-frame
             (dolist (window (window-list target-frame 'no-minibuf))
               (my/repair-window-after-buffer-kill window target-buffer)))))
       frame
       killed-buffer)))

  ;; Some kill paths do not go through `server-visit-hook' once a frame is
  ;; already alive, so repair bad fallback buffers after any buffer kill.
  (add-hook 'kill-buffer-hook
            (lambda ()
              (my/repair-buffer-fallback-after-kill (current-buffer))))
  (add-to-list 'default-frame-alist '(internal-border-width . 0)))
#+end_src

* User Config

The main configuration function, called after layers are loaded.  This is
where most of the custom behaviour lives.  Each subsection below maps to a
logical feature area.

** General settings

Whitespace display (trailing whitespace and tab marks), frame-title
suppression for a cleaner tiling WM experience, and a few embedded-focused
mode associations that Spacemacs does not cover out of the box.

#+begin_src emacs-lisp
(defun dotspacemacs/user-config ()
  (setq-default
   whitespace-style '(face trailing tabs tab-mark)
   whitespace-line-column 100)

  (setq frame-title-format nil
        indicate-buffer-boundaries nil
        indicate-empty-lines nil)

  ;; Which-Key: use a bottom side-window.  Keep keystroke echo in the echo
  ;; area (not in the which-key popup).  Remove the mode-line on the popup
  ;; window and grow by 1 line to avoid overlap with the echo area.
  (with-eval-after-load 'which-key
    (setq which-key-popup-type 'side-window
          which-key-side-window-location 'bottom
          which-key-show-prefix 'echo)
    ;; Hide the mode-line and grow by 1 line for clearance above echo area.
    (defun my/which-key-adjust-window (&rest _)
      (when-let ((win (get-buffer-window which-key--buffer)))
        (with-selected-window win
          (setq-local mode-line-format nil)
          (ignore-errors (enlarge-window 1))))
      (run-at-time 0 nil
                   (lambda ()
                     (when-let ((win (get-buffer-window which-key--buffer)))
                       (with-selected-window win
                         (setq-local mode-line-format nil)
                         (ignore-errors (enlarge-window 1)))))))
    (advice-add 'which-key--show-popup :after #'my/which-key-adjust-window))

  ;; Pixel-precise smooth scrolling (Emacs 29+).
  (when (fboundp 'pixel-scroll-precision-mode)
    (pixel-scroll-precision-mode 1))

  ;; Protect against accidentally opening huge/minified files.
  (when (fboundp 'global-so-long-mode)
    (global-so-long-mode 1))
#+end_src

** Tabs

Keep real editing buffers in a single Centaur Tabs group so switching between
files stays predictable.  Internal, generated, and transient buffers are left
out of the tab list entirely; terminals are the only visible non-file buffers.
When the current buffer is internal, keep showing the normal file tab row
instead of making the header line disappear.

#+begin_src emacs-lisp
  (with-eval-after-load 'centaur-tabs
    (defconst my/centaur-tabs-visible-special-buffer-regexp
      "\\`\\*\\(?:terminal\\|ghostel\\|shell\\|eshell\\|vterm\\)"
      "Special buffers that should remain visible in Centaur Tabs.")

    (defun my/centaur-tabs-layout-group-name ()
      "Return the tab group name for the current Spacemacs layout."
      (if (and (fboundp 'safe-persp-name) (fboundp 'get-frame-persp))
          (or (safe-persp-name (get-frame-persp)) "Workspace")
        "Workspace"))

    (defun my/centaur-tabs-buffer-groups ()
      "Put visible buffers in the active layout's tab group."
      (list (my/centaur-tabs-layout-group-name)))

    (defun my/centaur-tabs-buffer-name (buffer)
      "Return BUFFER's name whether BUFFER is a buffer object or a name."
      (if (bufferp buffer)
          (or (buffer-name buffer) "")
        (format "%s" buffer)))

    (defun my/centaur-tabs-file-buffer-p (buffer)
      "Return non-nil when BUFFER is backed by a visited file."
      (let ((buf (get-buffer buffer)))
        (when buf
          (with-current-buffer buf
            (or buffer-file-name
                (and (buffer-base-buffer)
                     (buffer-file-name (buffer-base-buffer))))))))

    (defun my/centaur-tabs-visible-special-buffer-p (name)
      "Return non-nil when special buffer NAME should be visible as a tab."
      (string-match-p my/centaur-tabs-visible-special-buffer-regexp name))

    (defun my/centaur-tabs-visible-buffer-p (buffer)
      "Return non-nil when BUFFER should be shown as a tab."
      (let ((name (my/centaur-tabs-buffer-name buffer)))
        (and (not (string-prefix-p " " name))
             (or (my/centaur-tabs-file-buffer-p buffer)
                 (my/centaur-tabs-visible-special-buffer-p name)))))

    (defun my/centaur-tabs-buffer-list ()
      "Return the buffers Centaur Tabs should present as tabs."
      (seq-filter #'my/centaur-tabs-visible-buffer-p
                  (if (and (fboundp 'get-current-persp)
                           (fboundp 'persp-buffers)
                           (fboundp 'persp-buffer-in-other-p*))
                      (let ((persp (get-current-persp)))
                        (if persp
                            (cl-copy-list (persp-buffers persp))
                          (seq-remove #'persp-buffer-in-other-p*
                                      (buffer-list))))
                    (buffer-list))))

    (defun my/centaur-tabs-hide-tab (buffer)
      "Hide the tab bar only for buffers where a header line is actively bad."
      (or (window-dedicated-p (selected-window))
          (string-prefix-p " " (my/centaur-tabs-buffer-name buffer))))

    (defun my/centaur-tabs-current-tabset ()
      "Return the active layout tabset even from internal buffers."
      (centaur-tabs-buffer-update-groups)
      (let ((tabset (centaur-tabs-get-tabset
                     (my/centaur-tabs-layout-group-name))))
        (when (my/centaur-tabs-visible-buffer-p (current-buffer))
          (centaur-tabs-select-tab-value (current-buffer) tabset))
        tabset))

    (setq centaur-tabs-buffer-groups-function #'my/centaur-tabs-buffer-groups
          centaur-tabs-buffer-list-function #'my/centaur-tabs-buffer-list
          centaur-tabs-current-tabset-function #'my/centaur-tabs-current-tabset
          centaur-tabs-hide-tab-function #'my/centaur-tabs-hide-tab)
    (clrhash centaur-tabs-hide-hash)
    (centaur-tabs-headline-match))
#+end_src

    "Configure CMake with exported compile commands in BUILD-DIR.

** Mode-line (spaceline)

Override the Spacemacs spaceline modeline faces with Gruvbox Dark Hard
palette colours.  We do NOT rebuild the spaceline theme — Spacemacs
already includes =mu4e-alert-segment=, =global=, and all standard
segments.  We only repaint the faces after each spaceline compilation.
The Spacemacs home buffer normally hides its mode-line with a buffer-local
empty value, so restore the stock Spaceline value there instead of creating
a separate home-only mode-line.  Keep the existing =mu4e-alert-segment=
high enough in Spaceline's responsive priority order that the mail count
survives normal narrow-window pruning.

#+begin_src emacs-lisp
  (defun my/spaceline-segment-spec-matches-p (spec segment)
    "Return non-nil when Spaceline SPEC contains SEGMENT."
    (or (eq spec segment)
        (and (consp spec)
             (or (eq (car spec) segment)
                 (and (consp (car spec))
                      (memq segment (car spec)))))))

  (defun my/spaceline-segment-index (segments segment)
    "Return the index of SEGMENT in Spaceline SEGMENTS."
    (let ((index 0)
          found)
      (while (and segments (not found))
        (when (my/spaceline-segment-spec-matches-p (car segments) segment)
          (setq found index))
        (setq segments (cdr segments)
              index (1+ index)))
      found))

  (defun my/prioritize-mu4e-alert-spaceline-segment (&rest _)
    "Keep Spaceline's existing mu4e-alert segment visible in normal widths."
    (when (and (boundp 'spaceline--mode-lines)
               (boundp 'spaceline--segments-code-main-left))
      (let* ((main (cdr (assq 'main spaceline--mode-lines)))
             (left-spec (car main))
             (index (my/spaceline-segment-index left-spec 'mu4e-alert-segment))
             (left-data (default-value 'spaceline--segments-code-main-left)))
        (when (and index (< index (length left-data)))
          (let ((runtime (nth index left-data)))
            (aset runtime 0 99)
            (aset runtime 2 t))
          (dolist (buffer (buffer-list))
            (with-current-buffer buffer
              (kill-local-variable 'spaceline--segments-code-main-left)
              (kill-local-variable 'spaceline--segments-code-main-right)
              (kill-local-variable 'spaceline--runtime-data-main)))))))

  (defun my/restore-spacemacs-home-spaceline (&rest _)
    "Use the regular Spaceline mode-line on the Spacemacs home buffer."
    (when (and (boundp 'spacemacs-buffer-name)
               (default-value 'mode-line-format))
      (let ((buffer (get-buffer spacemacs-buffer-name)))
        (when buffer
          (with-current-buffer buffer
            (setq-local mode-line-format (default-value 'mode-line-format))
            (force-mode-line-update))))))

  (with-eval-after-load 'core-spacemacs-buffer
    (unless (advice-member-p #'my/restore-spacemacs-home-spaceline
                             'spacemacs-buffer/goto-buffer)
      (advice-add 'spacemacs-buffer/goto-buffer
                  :after #'my/restore-spacemacs-home-spaceline))
    (unless (advice-member-p #'my/restore-spacemacs-home-spaceline

    "Return the current MPRIS service suffix for spotifyd or Spotify.

** mu4e stale-lock cleanup

When the mu server process dies unexpectedly it can leave the Xapian
database locked.  The advice below kills any orphaned =mu= processes
before (re)starting the server so mu4e recovers automatically.

#+begin_src emacs-lisp
  (defun my/mu4e-kill-stale-mu-processes ()
    "Kill any mu processes holding the Xapian database lock.
Called before starting the mu server so mu4e can acquire the lock."
    (let ((mu-pids
           (split-string
            (string-trim
             (shell-command-to-string
              (format "pgrep -u %s -x mu 2>/dev/null" (user-login-name))))
            "\n" t)))
      (when mu-pids
        (message "mu4e: found %d stale mu process(es), terminating: %s"
                 (length mu-pids) (string-join mu-pids ", "))
        (dolist (pid mu-pids)
          (let ((pid-num (string-to-number pid)))
            (when (> pid-num 0)
              (signal-process pid-num 15))))
        (sleep-for 1)
        (let ((remaining
               (split-string
                (string-trim
                 (shell-command-to-string
                  (format "pgrep -u %s -x mu 2>/dev/null" (user-login-name))))
                "\n" t)))
          (dolist (pid remaining)
            (let ((pid-num (string-to-number pid)))
              (when (> pid-num 0)
                (message "mu4e: force-killing stale mu process %d" pid-num)
                (signal-process pid-num 9))))
          (when remaining
            (sleep-for 0.5))))))

  (with-eval-after-load 'mu4e
    (when (fboundp 'mu4e--server-start)
      (advice-add 'mu4e--server-start :before
                  #'my/mu4e-kill-stale-mu-processes)))
#+end_src
